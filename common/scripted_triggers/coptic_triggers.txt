# Coptic culture, or any culture that diverged from or hybridized with it
coptic_culture_or_descendant_trigger = {
	OR = {
		culture = culture:coptic
		culture = { any_parent_culture_or_above = { this = culture:coptic } }
	}
}

# Vanilla Egyptian (Arabic) culture but a member of the Coptic Church (Coptic rite): a Copt
coptic_church_egyptian_trigger = {
	culture = culture:egyptian
	rite = rite:coptic_rite
}
