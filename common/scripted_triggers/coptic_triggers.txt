# The Coptic culture: the game's own since CK3 1.21 (key remenkimi, shown as "Remenkhemi"), any culture
# that keeps its Egyptian heritage (its divergences), and any culture formed directly from it (its hybrids).
# Arab Egyptian (culture:egyptian) lists the Coptic culture as a parent in the game files but is not Coptic.
coptic_culture_or_descendant_trigger = {
	culture = {
		OR = {
			has_cultural_pillar = heritage_egyptian
			this = culture:remenkimi
			AND = {
				NOT = { this = culture:egyptian }
				any_parent_culture = { this = culture:remenkimi }
			}
		}
	}
}

# A county of Egypt proper: the kingdom of Egypt without Cyrenaica, which was never Coptic land.
coptic_county_of_egypt_trigger = {
	kingdom ?= title:k_egypt
	NOT = { duchy ?= title:d_cyrenaica }
}

# A county of Egypt proper whose people speak Arabic.
coptic_arabic_county_of_egypt_trigger = {
	culture = { has_cultural_pillar = language_arabic }
	coptic_county_of_egypt_trigger = yes
}
